  <script>
    // Floating hearts background
    const heartEmojis = ["💗", "💕", "💖", "💘", "❤️", "🌸"];

    function createHeart() {
      const heart = document.createElement("span");
      heart.className = "floating-heart";
      heart.textContent =
        heartEmojis[Math.floor(Math.random() * heartEmojis.length)];

      heart.style.left = Math.random() * 100 + "vw";
      heart.style.fontSize = (16 + Math.random() * 20) + "px";
      heart.style.animationDuration = (5 + Math.random() * 5) + "s";

      document.body.appendChild(heart);
      heart.addEventListener("animationend", () => heart.remove());
    }

    const reducedMotion = window.matchMedia(
      "(prefers-reduced-motion: reduce)"
    );

    let heartTimer;

    function manageHearts() {
      clearInterval(heartTimer);
      if (!reducedMotion.matches) {
        heartTimer = setInterval(createHeart, 550);
      }
    }

    manageHearts();
    reducedMotion.addEventListener?.("change", manageHearts);

    // Responsive confetti canvas
    const canvas = document.getElementById("confetti");
    const ctx = canvas.getContext("2d");
    let pieces = [];
    let animationId = null;
    let lastFrame = 0;

    function resizeCanvas() {
      const dpr = Math.min(window.devicePixelRatio || 1, 2);
      canvas.width = window.innerWidth * dpr;
      canvas.height = window.innerHeight * dpr;
      canvas.style.width = window.innerWidth + "px";
      canvas.style.height = window.innerHeight + "px";
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    }

    window.addEventListener("resize", resizeCanvas);
    resizeCanvas();

    function celebrate() {
      const colors = [
        "#ff477e", "#ffb703", "#fb6f92",
        "#c77dff", "#ff85a1", "#ffd166",
        "#ffffff", "#e84378"
      ];

      const count = window.innerWidth < 500 ? 130 : 230;

      // Restart a previous celebration cleanly.
      if (animationId !== null) {
        cancelAnimationFrame(animationId);
      }

      pieces = Array.from({ length: count }, () => ({
        x: Math.random() * window.innerWidth,
        y: -Math.random() * window.innerHeight * 0.7,
        size: 4 + Math.random() * 7,
        speed: 2 + Math.random() * 4,
        drift: (Math.random() - 0.5) * 2.5,
        rotation: Math.random() * Math.PI * 2,
        spin: (Math.random() - 0.5) * 0.15,
        color: colors[Math.floor(Math.random() * colors.length)],
        shape: Math.random() > 0.5 ? "rect" : "circle"
      }));

      lastFrame = 0;
      animationId = requestAnimationFrame(drawConfetti);
    }

    function drawConfetti(timestamp) {
      // Cap frame rate to reduce unnecessary mobile rendering.
      if (timestamp - lastFrame < 1000 / 45) {
        animationId = requestAnimationFrame(drawConfetti);
        return;
      }
      lastFrame = timestamp;

      ctx.clearRect(0, 0, window.innerWidth, window.innerHeight);

      pieces.forEach(p => {
        p.y += p.speed;
        p.x += p.drift + Math.sin(p.y / 35) * 0.5;
        p.rotation += p.spin;

        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate(p.rotation);
        ctx.fillStyle = p.color;

        if (p.shape === "circle") {
          ctx.beginPath();
          ctx.arc(0, 0, p.size / 2, 0, Math.PI * 2);
          ctx.fill();
        } else {
          ctx.fillRect(
            -p.size / 2, -p.size / 2,
            p.size, p.size * 1.6
          );
        }

        ctx.restore();
      });

      pieces = pieces.filter(p => p.y < window.innerHeight + 30);

      if (pieces.length) {
        animationId = requestAnimationFrame(drawConfetti);
      } else {
        ctx.clearRect(0, 0, window.innerWidth, window.innerHeight);
        animationId = null;
      }
    }

    // Reveal the romantic message and celebrate.
    const button = document.getElementById("loveButton");
    const message = document.getElementById("loveMessage");

    button.addEventListener("click", () => {
      message.classList.add("show");
      button.textContent = "You're My Everything! 💖";

      if (!reducedMotion.matches) {
        celebrate();
      }
    });
  </script>
